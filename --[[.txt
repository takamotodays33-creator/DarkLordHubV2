local function getGuiParent()
    if gethui then
        local ok, hui = pcall(gethui)
        if ok and hui then return hui end
    end
    if syn and syn.protect_gui then
        local g = Instance.new("ScreenGui")
        pcall(function() syn.protect_gui(g) end)
        return g
    end
    if protect_gui then
        local g = Instance.new("ScreenGui")
        pcall(function() protect_gui(g) end)
        return g
    end
    return CoreGui
end

do
    local dragging, dragStart, startPos
    TitleBar.InputBegan:Connect(function(input)
        if input.UserInputType == Enum.UserInputType.MouseButton1
        or input.UserInputType == Enum.UserInputType.Touch then
            dragging = true
            dragStart = input.Position
            startPos = Main.Position
        end
    end)
    UserInputService.InputChanged:Connect(function(input)
        if dragging and (input.UserInputType == Enum.UserInputType.MouseMovement
        or input.UserInputType == Enum.UserInputType.Touch) then
            local delta = input.Position - dragStart
            Main.Position = UDim2.new(
                startPos.X.Scale, startPos.X.Offset + delta.X,
                startPos.Y.Scale, startPos.Y.Offset + delta.Y
            )
        end
    end)
    UserInputService.InputEnded:Connect(function(input)
        if input.UserInputType == Enum.UserInputType.MouseButton1
        or input.UserInputType == Enum.UserInputType.Touch then
            dragging = false
        end
    end)
end

    Instance.new("UICorner", barBg).CornerRadius = UDim.new(0, 3)

    fillGlow.Transparency = 0.5

    barBtn.Position = UDim2.new(0, 0, -1, 0)
    barBtn.BackgroundTransparency = 1
    barBtn.Parent = barBg

    local function updateFromInput(input)
        local relX = math.clamp((input.Position.X - barBg.AbsolutePosition.X) / barBg.AbsoluteSize.X, 0, 1)
        local val = math.floor(min + (max - min) * relX + 0.5)
        fill.Size = UDim2.new(relX, 0, 1, 0)
        valueLabel.Text = tostring(val)
        if callback then
            local ok, err = pcall(callback, val)
            if not ok then warn("[DarkLord] Slider: " .. tostring(err)) end
        end
    end

        if input.UserInputType == Enum.UserInputType.MouseButton1
        or input.UserInputType == Enum.UserInputType.Touch then
            dragging = true
            updateFromInput(input)
        end
    end)
    UserInputService.InputChanged:Connect(function(input)
        if dragging and (input.UserInputType == Enum.UserInputType.MouseMovement
        or input.UserInputType == Enum.UserInputType.Touch) then
            updateFromInput(input)
        end
    end)
    UserInputService.InputEnded:Connect(function(input)
        if input.UserInputType == Enum.UserInputType.MouseButton1
        or input.UserInputType == Enum.UserInputType.Touch then
            dragging = false
        end
    end)

    return row

local function switchTab(key)
    if activeTabKey == key then return end
    for k, page in pairs(Pages) do
        page.frame.Visible = (k == key)
    end

local function getTargets()
    local list = {}
    local myHRP = getHRP()
    if not myHRP then return list end
    local myPos = myHRP.Position
    for _, p in ipairs(Players:GetPlayers()) do
        if p ~= LP and p.Character then
            local hrp = p.Character:FindFirstChild("HumanoidRootPart")
            local hum = p.Character:FindFirstChildOfClass("Humanoid")
            if hrp and hum and hum.Health > 0 then
                table.insert(list, {
                    player = p,
                    hrp = hrp,
                    hum = hum,
                    distance = (hrp.Position - myPos).Magnitude,
                    health = hum.Health,
                })
            end
        end
    end
    return list
end

local function sortByNearest(list)
    table.sort(list, function(a, b) return a.distance < b.distance end)
    return list
end

local ESP_Boxes = {}

local function createESP(player)
    if ESP_Boxes[player] then return end
    local box = Instance.new("BoxHandleAdornment")
    box.Name = "DL_ESP"
    box.Adornee = nil
    box.AlwaysOnTop = true

    box.Color3 = FeatureState.espColor
    box.Parent = ScreenGui
    ESP_Boxes[player] = box
end

local function updateESP()
    for _, p in ipairs(Players:GetPlayers()) do
        if p == LP then continue end
        if not FeatureState.espEnabled then
            if ESP_Boxes[p] then ESP_Boxes[p]:Destroy(); ESP_Boxes[p] = nil end
            continue
        end
        if FeatureState.espTeamCheck and p.Team == LP.Team and LP.Team ~= nil then
            if ESP_Boxes[p] then ESP_Boxes[p].Adornee = nil end
            continue
        end
        if not ESP_Boxes[p] then createESP(p) end
        local hrp = p.Character and p.Character:FindFirstChild("HumanoidRootPart")
        ESP_Boxes[p].Adornee = hrp
        ESP_Boxes[p].Color3 = FeatureState.espColor
    end
end
